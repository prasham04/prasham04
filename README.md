on:
  schedule:
    - cron: "0 0 * * *"

  workflow_dispatch:

permissions:
  contents: write

jobs:

  generate:

    runs-on: ubuntu-latest

    timeout-minutes: 10

    steps:

      - name: Generate contribution snake

        env:

          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
